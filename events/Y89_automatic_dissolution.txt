add_namespace = Y89_breakup

country_event = {
    id = Y89_breakup.1
    title = Y89_breakup.1.t
    desc = Y89_breakup.1.d
    picture = GFX_report_event_generic_parliament
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        Y89_historical_breakup_allowed = yes
        date > 1991.6.24
        NOT = { has_country_flag = Y89_SLOVENIA_CROATIA_SEPARATED }
    }
    immediate = {
        # The console can bypass event triggers; guard the effects as well.
        if = {
            limit = {
                Y89_historical_breakup_allowed = yes
                date > 1991.6.24
                NOT = { has_country_flag = Y89_SLOVENIA_CROATIA_SEPARATED }
            }
            set_country_flag = Y89_SLOVENIA_CROATIA_SEPARATED
            if = {
                limit = { NOT = { country_exists = SLV } }
                release = SLV
                SLV = {
                    if = { limit = { exists = yes num_divisions < 1 } load_oob = SLV_1936 }
                }
            }
            complete_national_focus = Y89_slovenia_independence
            if = {
                limit = { NOT = { country_exists = CRO } }
                release = CRO
                CRO = {
                    if = { limit = { exists = yes num_divisions < 1 } load_oob = CRO_1936 }
                }
            }
            complete_national_focus = Y89_croatia_independence
        }
    }
    option = { name = Y89_breakup.1.a }
}
